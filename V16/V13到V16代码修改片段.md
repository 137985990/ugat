### 替换为：

```python
# 创建主生成模型（TGATUNet）
model = TGATUNet(
    in_channels=input_channels,
    hidden_channels=config.get('hidden_channels', 64),
    out_channels=input_channels,
    num_classes=config.get('num_classes', 2)
).to(device)
logger.info(f"主生成模型创建完成，参数量: {sum(p.numel() for p in model.parameters()):,}")

# 创建第1个分类器：用于评估输入数据（baseline性能）
input_classifier = nn.Sequential(
    nn.Linear(input_channels, config.get('hidden_channels', 64)),
    nn.ReLU(),
    nn.Dropout(0.1),
    nn.Linear(config.get('hidden_channels', 64), config.get('num_classes', 2))
).to(device)
logger.info(f"输入数据分类器创建完成，参数量: {sum(p.numel() for p in input_classifier.parameters()):,}")

# 创建第2个分类器：用于评估重建数据（目标性能）
reconstructed_classifier = nn.Sequential(
    nn.Linear(input_channels, config.get('hidden_channels', 64)),
    nn.ReLU(),
    nn.Dropout(0.1),
    nn.Linear(config.get('hidden_channels', 64), config.get('num_classes', 2))
).to(device)
logger.info(f"重建数据分类器创建完成，参数量: {sum(p.numel() for p in reconstructed_classifier.parameters()):,}")

# 合并所有模型的参数进行联合优化
all_parameters = list(model.parameters()) + list(input_classifier.parameters()) + list(reconstructed_classifier.parameters())
optimizer = Adam(all_parameters, lr=config.get('learning_rate', 1e-4), weight_decay=config.get('weight_decay', 1e-5))
```

## 修改片段4：分类计算逻辑（train.py）

### 在训练函数中找到分类计算部分，替换为：
```python
                # 阶段2: 双路径分类训练 - 使用两个独立分类器进行评估
                # 路径1: 输入数据分类 (Common真+Have真+Need初始/上轮生成) → 独立分类器1 → 基准性能
                input_data_pooled = batch.mean(dim=2)  # [batch_size, input_channels] 时间维度池化
                if input_classifier is not None:
                    input_logits = input_classifier(input_data_pooled)
                else:
                    _, input_logits = forward_batch_parallel(model, batch, device)
                
                # 路径2: 重建数据分类 (Common真+Have真+Need当前生成) → 独立分类器2 → 目标性能
                # 构建增强数据：保留Common+Have，用当前模型生成的Need替换
                enhanced_data = batch.clone()
                if need_indices:
                    for need_idx in need_indices:
                        if need_idx < batch_reconstructed.size(1):
                            enhanced_data[:, need_idx, :] = batch_reconstructed[:, need_idx, :]
                
                enhanced_data_pooled = enhanced_data.mean(dim=2)  # [batch_size, input_channels] 时间维度池化
                if reconstructed_classifier is not None:
                    enhanced_logits = reconstructed_classifier(enhanced_data_pooled)
                else:
                    _, enhanced_logits = forward_batch_parallel(model, enhanced_data, device)
```
